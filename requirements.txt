# Pins are exact so a clone in three weeks builds the system we tested, not whatever
# resolved that morning. Behavior 6 is "fresh clone to working system"; a floating
# pin is how that promise quietly stops being true.
#
# Every version below was resolved against PyPI and installed successfully on
# 2026-08-15 — none are from memory.

fastapi==0.141.1

pydantic-settings==2.15.0

# Required by FastAPI to parse multipart bodies — the document upload endpoint.
# FastAPI does not depend on it, and without it the route raises at import time
# rather than on first request.
python-multipart==0.0.32

# psycopg MUST carry the [binary] extra.
#
# Plain `psycopg` installs cleanly and then fails at *import* with "no pq wrapper
# available", because it expects a system libpq that Windows does not have and slim
# container images do not ship. The failure surfaces inside LangGraph's PostgresSaver,
# which makes it look like a LangGraph bug and sends you debugging the wrong library.
# The [binary] extra bundles a compiled libpq. Verified on this machine, 2026-08-15.
psycopg[binary]==3.3.4

SQLAlchemy==2.0.52
alembic==1.19.1
pgvector==0.5.0

langgraph==1.2.11
langgraph-checkpoint-postgres==3.1.2

httpx==0.28.1
PyYAML==6.0.3
pypdf==6.4.0

# The MCP surface (mcp_server.py). Behavior 4's strongest shape, and the one SuperDocs
# uses itself. 2.x renamed a good deal from 1.x — `MCPServer` replaces `FastMCP`,
# fields are snake_case, and tool failures are `ToolError` while `MCPError` tears down
# the session — so the major version here is load-bearing, not cosmetic.
